    Dim ObjectName As String
    Dim Customer As String
    Dim LineToWrite As String

    ' Укажите путь к файлу
    FilePath = "D:\Temp\Закрытие.txt"

    ' Получение текущей даты и времени
    CurrentDateTime = Now()

    ' Чтение значений из именованных диапазонов
    On Error Resume Next ' Обработка ошибок
    SummaKS2 = ThisWorkbook.Names("Сумма_КС2").RefersToRange.Value
    ObjectName = ThisWorkbook.Names("ИмяОБЪЕКТАвРЕЕСТР").RefersToRange.Value
    Customer = ThisWorkbook.Names("Заказчик").RefersToRange.Value
    If Err.Number <> 0 Then
        MsgBox "Один или несколько именованных диапазонов не найдены!", vbCritical
        Exit Sub
    End If
    On Error GoTo 0

    ' Формирование строки для записи
    LineToWrite = CurrentDateTime & vbTab & "КС2 на сумму:" & SummaKS2 & vbTab & ObjectName & vbTab & Customer

    ' Открытие файла для добавления данных
    FileNum = FreeFile
    Open FilePath For Append As #FileNum

    ' Запись строки в файл
    Print #FileNum, LineToWrite

    ' Закрытие файла
    Close #FileNum
    MsgBox "Данные сохранены в D:\Temp\Закрытие.txt", vbInformation

End Sub
